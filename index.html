// --- Gemini API 呼び出し（タイムアウト機能 ＆ 自動リトライ付き） --- async function callGeminiAPI(userPrompt, maxRetries = 3) { const apiKey = loadApiKey(); if (!apiKey) { dbgLog("[API Error] APIキーが未設定です", true); return { emotion: 'Sad', text: "APIキーが設定されていないみたい。" }; } const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3.8-flash:generateContent?key=${apiKey}`; const systemInstruction = `あなたは可愛らしいAIペットです。ユーザーの発言に対して、以下のJSON形式でのみ答えてください。他の余計な文やコードブロック指定（\`\`\`jsonなど）は含めないでください。 フォーマット: {"emotion":"Happy|Anger|Surprised|Sad|Default","text":"40文字以内の話し言葉"} 感情基準: Happy: 嬉しい、挨拶、褒められた Anger: 怒られた、いじわる Surprised: 驚き、びっくり Sad: 悲しい、バイバイ Default: その他通常会話`; const payload = { contents: [ { role: "user", parts: [ { text: `${systemInstruction}\n\nユーザーの発言: ${userPrompt}` } ] } ] }; for (let attempt = 1; attempt <= maxRetries; attempt++) { // ★ 7秒で強制遮断する AbortController const controller = new AbortController(); const timeoutId = setTimeout(() => controller.abort(), 7000); try { dbgLog(`[Gemini リクエスト送信 (試行 ${attempt}/${maxRetries})] "${userPrompt}"`); const response = await fetch(endpoint, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload), signal: controller.signal // タイムアウト制御用シグナル }); clearTimeout(timeoutId); // 通信成功したらタイマー解除 if (!response.ok) { const errData = await response.json().catch(() => ({})); const errMsg = errData.error?.message || `HTTP ${response.status}`; throw new Error(errMsg); } const data = await response.json(); const rawText = data.candidates?.[0]?.content?.parts?.[0]?.text?.trim() || ""; dbgLog(`[Gemini 受信テキスト] ${rawText}`); const jsonMatch = rawText.match(/\{[\s\S]*?\}/); if (jsonMatch) { const parsed = JSON.parse(jsonMatch[0]); const validEmotions = ['Happy', 'Anger', 'Surprised', 'Sad', 'Default']; const emotion = validEmotions.includes(parsed.emotion) ? parsed.emotion : 'Happy'; const text = parsed.text || "うん！分かったよ！"; dbgLog(`[Gemini 解析成功] 感情:${emotion}, テキスト:"${text}"`); return { emotion, text }; } else { dbgLog("[Gemini 警告] JSON構造が見つかりません。直接テキストとして処理します", true); return { emotion: 'Happy', text: rawText.substring(0, 40) }; } } catch (err) { clearTimeout(timeoutId); const isAbort = err.name === 'AbortError'; const errorReason = isAbort ? "通信タイムアウト (7秒制限)" : err.message; dbgLog(`[Gemini 試行 ${attempt} 失敗] ${errorReason}`, true); if (attempt === maxRetries) { return { emotion: 'Sad', text: "ごめんね、通信が混雑していて上手くおえんじが考えられなかったよ。" }; } const delayMs = Math.pow(2, attempt - 1) * 1000; dbgLog(`[Gemini 自動リトライ] ${delayMs / 1000}秒後に再試行します...`); await new Promise(resolve => setTimeout(resolve, delayMs)); } } }
